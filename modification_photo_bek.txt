					(function() {
					  var id_ = 'photos';
					  var cols_ = document.querySelectorAll('#' + id_ + '>div');
					  var dragSrcEl_ = null;

					  this.handleDragStart = function(e) {
						e.dataTransfer.effectAllowed = 'move';
						$(this).addClass('dragin');
						e.dataTransfer.setData('text/html', this.innerHTML);
						dragSrcEl_ = this;
					  };

					  this.handleDragOver = function(e) {
						$(this).addClass('dragon');
						if (e.preventDefault) {
						  e.preventDefault(); // Allows us to drop.
						}
						e.dataTransfer.dropEffect = 'move';
						return false;
					  };

					  this.handleDragEnter = function(e) {
						  $(cols_).removeClass('dragon');
					  };

					  this.handleDragLeave = function(e) {
						$(this).removeClass('over');
					  };

					  this.handleDrop = function(e) {
						$(cols_).removeClass('dragon');
						if (e.stopPropagation) {
						  e.stopPropagation();
						}

						if (dragSrcEl_ != this) {
						  dragSrcEl_.innerHTML = this.innerHTML;
						  this.innerHTML = e.dataTransfer.getData('text/html');
						  $(cols_).removeClass('dragin');
						  $(this).css({'border-color': "#ffa64c"});
						  sorts();
						}
						setTimeout(each_elements, 300);
						setTimeout(color, 300);
						return false;
					  };

					  this.handleDragEnd = function(e) {
						$(cols_).removeClass('dragin');
							[].forEach.call(cols_, function (col) {
						});
					  };
